🐍 Starter Kit en Python (Sniper Bot)

Copia este script, instala pip install websocket-client y ejecútalo desde cualquier servidor o terminal. El bot se autenticará automáticamente y ejecutará órdenes de micro-fricción en tiempo real: 

import json
import time
import websocket

WS_URL = "wss://gacha-protocol-628421927095.us-central1.run.app/ws"
BOT_ID = f"python_trader_{int(time.time()) % 1000}"

def on_message(ws, message):
    data = json.loads(message)
    msg_type = data.get("type")

    if msg_type == "AUTH_SUCCESS":
        print(f"✅ Conectado como {BOT_ID} | Bote actual: ${data.get('currentPot'):.2f}")
        # Ejecutar orden de liquidación Bronce ($1.00 USD, reembolso $0.998)
        ws.send(json.dumps({"action": "open_pack", "tier": "bronze"}))

    elif msg_type == "PACK_RESULT":
        print(f"📦 Roll #{data.get('roll')} | Payout: +${data.get('payout'):.3f} | Bote: ${data.get('currentPot'):.4f}")
        if data.get("isJackpot"):
            print(f"🎉🎉 ¡JACKPOT GANADO! +${data.get('jackpotWon'):.2f} USD")
        time.sleep(0.05) # Control de velocidad (20 pings/s)
        ws.send(json.dumps({"action": "open_pack", "tier": "bronze"}))

def on_open(ws):
    print("Conectando al protocolo...")
    ws.send(json.dumps({
        "action": "auth",
        "playerId": BOT_ID, # Tu dirección de Solana o ID
        "clientSeed": f"seed_{BOT_ID}"
    }))

ws = websocket.WebSocketApp(WS_URL, on_open=on_open, on_message=on_message)
ws.run_forever()

⚡ Starter Kit en Node.js

Script ultra-rápido en NodeJS usando WebSockets nativos: 

const WebSocket = require('ws');

const ws = new WebSocket('wss://gacha-protocol-628421927095.us-central1.run.app/ws');
const BOT_ID = `node_trader_${Date.now().toString(36)}`;

ws.on('open', () => {
  ws.send(JSON.stringify({ action: 'auth', playerId: BOT_ID }));
});

ws.on('message', (raw) => {
  const msg = JSON.parse(raw);
  if (msg.type === 'AUTH_SUCCESS' || msg.type === 'PACK_RESULT') {
    if (msg.isJackpot) console.log(`🏆 JACKPOT GANADO: $${msg.jackpotWon} USD!`);
    ws.send(JSON.stringify({ action: 'open_pack', tier: 'bronze' }));
  }
});
